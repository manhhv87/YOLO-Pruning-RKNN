% CALM-Row pipeline overview (TikZ; no external assets).
\begin{figure}[!t]
  \centering
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}[
    node distance=7mm and 8mm,
    box/.style={draw, rounded corners=2pt, align=center, font=\small,
                inner sep=4pt, minimum height=13mm, text width=25mm},
    npu/.style={box, fill=blue!8, draw=blue!50},
    cpu/.style={box, fill=orange!12, draw=orange!60!black},
    io/.style={box, fill=black!5, text width=18mm},
    flow/.style={-{Latex[length=2.4mm]}, thick},
    loss/.style={-{Latex[length=2.2mm]}, thick, dashed, red!70!black},
  ]
    % ---- main forward flow (top row) ----
    \node[io]  (img)                    {Input frame};
    \node[npu, right=of img]  (det)     {Stock YOLO\\backbone\,+\,neck\\+\,DFL head\\\scriptsize\cite{ct4,yolov10,gfl}};
    \node[npu, right=of det]  (dist)    {Per-edge box\\distributions\\$p_e(k)$};
    \node[cpu, right=of dist] (var)     {Edge moments\\$\mu_e,\ \sigma_e^2$};
    \node[cpu, right=of var]  (cov)     {Centroid cov.\\$\sigma_{c_x}^2{=}\tfrac{s^2}{4}(\sigma_l^2{+}\sigma_r^2)$};

    % ---- second row: fit + output ----
    \node[cpu, below=14mm of cov, text width=34mm] (gls)
        {Precision-weighted GLS line\\$x{=}ay{+}b,\quad w_i{=}1/\sigma_{c_x,i}^2$};
    \node[io, left=14mm of gls, text width=24mm] (out)
        {Guidance line\\heading $\theta{=}\arctan a\ {\pm}\ \sigma_\theta$};

    \draw[flow] (img)--(det);
    \draw[flow] (det)--(dist);
    \draw[flow] (dist)--(var);
    \draw[flow] (var)--(cov);
    \draw[flow] (cov)--(gls);
    \draw[flow] (gls)--(out);

    % ---- training-only feedback (losses) ----
    \draw[loss] (gls.west) to[bend right=22]
        node[midway, below, font=\scriptsize\itshape, align=center]{$L_\mathrm{LSL}$\\(line stability)} (det.south);
    \draw[loss] (cov.north) to[bend right=28]
        node[midway, above, font=\scriptsize\itshape, align=center]{$L_\mathrm{DDC}$ (calibration)} (dist.north);

    % ---- grouping boxes (NPU vs CPU) ----
    \begin{scope}[on background layer]
      \node[draw=blue!50, dashed, rounded corners, fit=(det)(dist),
            inner sep=5mm, label={[blue!50!black,font=\scriptsize]above:On-device NPU graph (INT8 RKNN / TensorRT)\,---\,vanilla YOLO}] {};
      \node[draw=orange!60!black, dashed, rounded corners, fit=(var)(cov)(gls)(out),
            inner sep=5mm, label={[orange!60!black,font=\scriptsize]below:CPU, post-NMS\,---\,CALM-Row (no new params / no NPU ops)}] {};
    \end{scope}
  \end{tikzpicture}}
  \caption{CALM-Row pipeline. A \emph{stock} YOLO detector (blue) emits, per box edge $e\in\{l,t,r,b\}$,
  a Distribution-Focal-Loss distribution $p_e(k)$~\cite{gfl}; its mean is the usual box coordinate while
  its \emph{variance} $\sigma_e^2$ is read out (orange, CPU, post-NMS) and propagated to a per-box centroid
  covariance, then used as the precision weight $w_i$ in a differentiable generalized-least-squares fit of the
  guidance line $x=ay+b$ and its heading $\theta$. Two training-only losses (dashed red) make the variance
  \emph{calibrated} ($L_\mathrm{DDC}$) and the fitted line \emph{stable} ($L_\mathrm{LSL}$), backpropagating
  through the weights into the existing edge logits. No new learnable parameters and no new operators on the
  feature maps are introduced, so the exported edge graph stays a vanilla YOLO.}
  \label{fig:pipeline}
\end{figure}
